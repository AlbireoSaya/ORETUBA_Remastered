function RVMExtraMem35(rd, wr, ctx, p) {
    var sp = rd((ctx + 12) & 0xffffffff) & 0xffffffff;
    var off = rd(p & 0xffffffff) & 0xffffffff;
    var src = (sp - off) & 0xffffffff;
    var dest = rd((p + 4) & 0xffffffff) & 0xffffffff;
    var value = rd(src) & 0xffffffff;
    wr(dest & 0xffffffff, value & 0xffffffff);
    var pc = rd((ctx + 8) & 0xffffffff) & 0xffffffff;
    wr((ctx + 8) & 0xffffffff, (pc + 8) & 0xffffffff);
}
