function RValueExtraCompare(op, a, b)
{
	if (typeof op != "Integer" || typeof a != "Integer" || typeof b != "Integer")
		throw new global.Exception("RValueExtraCompare: op/a/b must be Integer");

	var ua = a & 0xffffffff;
	var ub = b & 0xffffffff;

	var sa = ua;
	if (sa >= 0x80000000)
		sa = sa - 0x100000000;

	var sb = ub;
	if (sb >= 0x80000000)
		sb = sb - 0x100000000;

	if (op == 42)
	{
		if (sa <= sb)
			return 1;
		return 0;
	}

	if (op == 43)
	{
		if (sa > sb)
			return 1;
		return 0;
	}

	if (op == 46)
	{
		if (ua != ub)
			return 1;
		return 0;
	}

	throw new global.Exception("RValueExtraCompare: unknown op " + op);
}
