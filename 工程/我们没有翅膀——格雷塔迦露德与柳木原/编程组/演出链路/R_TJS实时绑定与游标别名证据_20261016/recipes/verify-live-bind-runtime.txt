"""Root platform engine invocation and exact editor-fixture count verification."""
from pathlib import Path
import hashlib
import json
import shutil
import sys
import time
import uuid

root = Path.cwd().resolve()
group = root / "工程/我们没有翅膀——格雷塔迦露德与柳木原/编程组"
work = root / "子代理工作区/ISSUE-20261011-02/r-target-live-bind-20261016"
sys.path.insert(0,str(group/"工具"))
from performance_runtime import execute

bits,tag,manifest_path=int(sys.argv[1]),sys.argv[2],Path(sys.argv[3]).resolve()
installed=len(sys.argv)>4 and sys.argv[4]=="installed"
override=Path(sys.argv[4]).resolve() if len(sys.argv)>4 and not installed else None
assert override is None or override.is_relative_to(root/"临时/ISSUE-20261011-02")
assert bits in (32,64) and "/" not in tag and "\\" not in tag and manifest_path.is_relative_to(root)
manifest=json.loads(manifest_path.read_text(encoding="utf-8-sig"))
assert 1<=manifest["cases"]<=1000
reference=manifest_path.parent/"original-results.json"
if not reference.exists():
    reference=root/manifest["original_result_source"]
assert reference.resolve().is_relative_to(root)
assert hashlib.sha256(reference.read_bytes()).hexdigest()==manifest["original_result_sha256"] or hashlib.sha256(reference.read_bytes().replace(b"\r\n",b"\n")).hexdigest()==manifest.get("original_result_sha256_lf")
assert json.loads(reference.read_text(encoding="utf-8-sig"))["passed"]
out=root/"临时/ISSUE-20261011-02"/tag
assert not out.exists()
game,save,evidence=out/"game",out/"savedata",out/"evidence"
for path in (game,save,evidence):path.mkdir(parents=True)
memory=["r_memory_region.tjs","r_memory_create.tjs","r_memory_access.tjs","r_memory_find.tjs","r_memory_read.tjs","r_memory_write8.tjs","r_memory_write32.tjs","r_memory_read16.tjs"]
stage=group/"演出链路/R_TJS实时绑定与游标别名证据_20261016"
delivery=json.loads((stage/"delivery-inventory.json").read_text(encoding="utf-8")) if installed else []
installed_sources={row["name"]:root/row["destination"]for row in delivery}
installed_hashes={row["name"]:row["sha256_lf"]for row in delivery}
owned=list(installed_sources) if installed else list(json.loads((work/"accepted-map.json").read_text(encoding="utf-8-sig")))
assert len(owned)==13
names=memory+owned+["r_vm_fixture_compare.tjs"]
assert len(names)==len(set(names))
sources=[]
for name in names:
    src=group/"原作忠实层/R"/name if name in memory else work/name if name in owned else group/"演出链路/R_TJS表达式VM与暂停恢复证据_20261015"/name
    if installed and name in installed_sources:src=installed_sources[name]
    if installed and name in installed_sources:assert hashlib.sha256(src.read_bytes().replace(b"\r\n",b"\n")).hexdigest()==installed_hashes[name]
    if override is not None and (override/name).exists():src=override/name
    shutil.copyfile(src,game/name)
    sources.append({"name":name,"source":src.relative_to(root).as_posix(),"sha256":hashlib.sha256(src.read_bytes()).hexdigest()})
nonce=uuid.uuid4().hex
commands=["Debug.startLogToFile();","try {",f'Debug.message("RLIVE_BEGIN {nonce}");']
for name in names:
    commands.extend((f'Debug.message("RLIVE_LOAD {nonce} {name}");',f'Scripts.execStorage("{name}");'))
for i,row in enumerate(manifest["files"]):
    src=root/row["source"]
    assert src.resolve().is_relative_to(root)
    assert hashlib.sha256(src.read_bytes()).hexdigest()==row["sha256"] or hashlib.sha256(src.read_bytes().replace(b"\r\n",b"\n")).hexdigest()==row.get("sha256_lf")
    dest=f"cases-{i:04d}.tjs"
    shutil.copyfile(src,game/dest)
    commands.append(f'Scripts.execStorage("{dest}");')
commands.extend((f'if(RLiveBindCaseCount!=={manifest["cases"]})throw new Exception("WRONG_LIVE_BIND_COUNT "+RLiveBindCaseCount);',
                 f'Debug.message("RLIVE_PASS {nonce} "+RLiveBindCaseCount);',"} catch(e) {",
                 f'Debug.message("RLIVE_EXCEPTION {nonce} "+((typeof e=="Object")?e.message:(""+e)));',"}","System.exit();"))
(game/"startup.tjs").write_text("\n".join(commands)+"\n",encoding="utf-8")
engine=root/f"工具/krkrz/krkrz_20171225/tvpwin{bits}.exe"
start=time.time_ns()
process=execute([str(engine),str(game),"-datapath="+str(save)+"/","-readencoding=UTF-8","-startup=startup.tjs","-userconf=no"],out,45,evidence)
log=save/"krkr.console.log"
raw=log.read_bytes() if log.exists() else b""
console=raw.decode("utf-16") if raw.startswith(b"\xff\xfe") else raw.decode("utf-16-le",errors="replace")
fresh=log.exists() and log.stat().st_mtime_ns>=start
passed=process["returncode"]==0 and not process["timed_out"] and process["cleanup_ok"] and fresh and f'RLIVE_PASS {nonce} {manifest["cases"]}' in console and f"RLIVE_EXCEPTION {nonce}" not in console
receipt={"passed":passed,"bits":bits,"cases":manifest["cases"],"scoped_original_events":manifest["events"],"fresh_log":fresh,"nonce":nonce,"process":process,
         "sources":sources,"input_manifest":manifest_path.relative_to(root).as_posix(),"original_result_sha256":manifest["original_result_sha256"],
         "engine_sha256":hashlib.sha256(engine.read_bytes()).hexdigest(),"console_sha256":hashlib.sha256(raw).hexdigest(),
         "scope":"exact scoped read/write events, cursor cell, 512 source final bytes, written final words; existing-bank fixture only; no VM, growth or full game proof"}
(evidence/"verdict.json").write_text(json.dumps(receipt,ensure_ascii=False,indent=2)+"\n",encoding="utf-8")
(evidence/"console.utf8.txt").write_text(console,encoding="utf-8")
print(json.dumps({k:receipt[k]for k in("passed","bits","cases","scoped_original_events","process")}))
print("\n".join(line for line in console.splitlines()if "RLIVE_"in line)[-1600:])
sys.exit(0 if passed else 1)
