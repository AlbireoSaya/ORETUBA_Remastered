u32 = lambda n: struct.pack("<I", n & 0xffffffff)

def table_bytes(groups, compact):
    data = bytearray()
    counts, indices = [], {}
    for group_id, rows in enumerate(groups):
        counts.append(len(data))
        data += bytes([len(rows)]) if compact & 255 else u32(len(rows))
        for i, (slot, index) in enumerate(rows):
            data += u32(slot)
            indices[group_id, i] = len(data)
            data += u32(index)
    return data, counts, indices

def blank():
    return [[] for _ in range(5)]

def direct(label, groups, compact=1, string_base=0x10d0000, code=normal_code, cursor=normal_cursor):
    payload, _, _ = table_bytes(groups, compact)
    data = bytearray(b"\xcc" * 512)
    data[table-source:table-source+len(payload)] = payload
    cases.append({"label": label, "kind": "direct", "compact": compact, "code": code, "table": table, "cursor": cursor, "string_base": string_base, "data": data})

def wrapper(label, groups, flag=0, code_size=64, declared_size=None):
    payload, _, _ = table_bytes(groups, 1)
    size = 2 + len(payload) if declared_size is None else declared_size
    expression = struct.pack("<H", code_size) + b"\0" * code_size + bytes([flag]) + struct.pack("<H", size) + payload
    body = b"\1\x07\x03" + expression
    record = b"[ " + u32(4) + b"\1" + u32(4+len(body)) + body
    cases.append({"label": label, "kind": "wrapper", "compact": 1, "code": source+16, "expression": source+14,
                  "cursor": None, "string_base": 0x10d0000, "data": bytearray(record + b"\xcc"*(512-len(record)))})
