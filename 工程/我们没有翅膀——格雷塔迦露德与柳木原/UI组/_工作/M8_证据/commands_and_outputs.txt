================================================================
$ rm -rf "$RUN/game"; rm -f "$RUN"/runtime/savedata/*
$ cp -r "$WT/krkr/game" "$RUN/game"
$ cp -r "$WT/krkr/assets" "$RUN/game/assets"
$ printf 'autotour' > "$RUN/game/autotour.txt"
$ cd "$RUN/runtime" && ./tvpwin32.exe "$RUN/game" > /dev/null 2>&1 &
$ sleep 55          # 约 45 秒巡演完并自行退出
$ taskkill //IM tvpwin32.exe //F      # 只杀自己启动的；正常退出后此命令报"没有找到进程"
